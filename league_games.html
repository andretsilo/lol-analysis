Introduction

League of Legends (LoL) is a multiplayer online battle arena (MOBA) developed by Riot Games. Two teams of five players compete on a map called Summoner’s Rift, each starting from opposite corners. The goal is to destroy the enemy team’s Nexus, the core structure at the heart of their base.

Matches typically last 25–35 minutes and revolve around three pillars: gold, objectives, and kills.

Gold is the primary resource. Players earn it by killing minions (CS), destroying structures, and eliminating enemy champions. Gold is spent on items that increase a champion’s power, making it the main driver of in-game strength.

Kills eliminate enemy champions, denying them gold and experience while rewarding the killer. Beyond the direct advantage, kills often open windows to take objectives or push structures.

Objectives are neutral monsters and structures that provide team-wide buffs:

  • Dragon: spawns in the river. Each kill grants a stacking elemental buff. Killing four grants Drake Soul, a permanent powerful bonus.
  • Elder Dragon: spawns after a team has Drake Soul. Killing it grants a devastating temporary buff that can decide teamfights.
  • Baron Nashor: the most powerful neutral monster. Killing it empowers the team’s minions for 3 minutes, enabling a strong push to end the game.
  • Rift Herald: an early-game objective that drops an item used to destroy turrets, accelerating map control.

load_matches <- function(folder_path, max_files = NULL) {
  json_files <- list.files(folder_path, pattern = "\\.json$", full.names = TRUE, recursive = TRUE)
  if (length(json_files) == 0) stop("No JSON files found in: ", folder_path)
  if (!is.null(max_files)) json_files <- head(json_files, max_files)

  n <- length(json_files)
  raw <- vector("list", n)
  for (i in seq_len(n)) {
    raw[[i]] <- tryCatch(fromJSON(json_files[i], simplifyVector = FALSE), error = function(e) NULL)
  }
  raw <- compact(raw)

  participants_list <- unlist(lapply(raw, \(d) {
    matchId <- d$metadata$matchId
    lapply(d$info$participants, \(p) list(
      matchId                     = matchId,
      gameId                      = d$info$gameId,
      gameDuration                = d$info$gameDuration,
      gameMode                    = d$info$gameMode,
      participantId               = p$participantId,
      puuid                       = p$puuid,
      summonerName                = p$riotIdGameName,
      championName                = p$championName,
      teamId                      = p$teamId,
      teamPosition                = p$teamPosition,
      win                         = p$win,
      kills                       = p$kills,
      deaths                      = p$deaths,
      assists                     = p$assists,
      champLevel                  = p$champLevel,
      totalDamageDealtToChampions = p$totalDamageDealtToChampions,
      goldEarned                  = p$goldEarned,
      totalMinionsKilled          = p$totalMinionsKilled,
      visionScore                 = p$visionScore,
      summonerLevel               = p$summonerLevel
    ))
  }), recursive = FALSE)
  df <- bind_rows(participants_list)

  bans <- bind_rows(lapply(raw, \(d) {
    matchId <- d$metadata$matchId
    bind_rows(lapply(d$info$teams, \(t)
      bind_rows(lapply(t$bans, \(b) list(championId = b$championId, pickTurn = b$pickTurn)))
    )) |> mutate(matchId = matchId)
  }))

  team_objectives <- bind_rows(lapply(raw, \(d) {
    matchId <- d$metadata$matchId
    bind_rows(lapply(d$info$teams, \(t) {
      team_participants <- d$info$participants[sapply(d$info$participants, \(p) p$teamId) == t$teamId]
      list(
        matchId     = matchId,
        teamId      = t$teamId,
        win         = t$win,
        dragonKills = t$objectives$dragon$kills,
        baronKills  = t$objectives$baron$kills,
        heraldKills = t$objectives$riftHerald$kills,
        elderKills  = max(sapply(team_participants, \(p) as.integer(p$challenges$teamElderDragonKills %||% 0L)))
      )
    }))
  }))

  team_gold <- df |> summarise(teamGold = sum(goldEarned), .by = c(matchId, teamId))

  result <- df |>
    left_join(team_gold, by = c("matchId", "teamId")) |>
    left_join(rename(team_gold, enemyTeamId = teamId, enemyGold = teamGold), by = "matchId", relationship = "many-to-many") |>
    filter(teamId != enemyTeamId) |>
    mutate(goldDiff = (teamGold - enemyGold) / 1000) |>
    select(-enemyTeamId, -enemyGold, -teamGold)

  list(matches = result, bans = bans, team_objectives = team_objectives)
}

Exploratory Data Analysis

The dataset does not include per-player rank/elo. Summoner level is used as a proxy for player experience. Game duration is in seconds and converted to minutes.

Game Duration

Most ranked games cluster between 25 and 35 minutes. Short games (under 20 min) typically indicate early surrenders, while games beyond 40 minutes are late-game stalemates.

chart <- matches |>

  ggplot(aes(x = minutes)) +

  theme_minimal(base_size = 14) +
  labs(title = "Game Duration Distribution", x = "Duration (minutes)", y = "Count")

Key Variables by Outcome

ECDF of gold, vision, and kills difference split by match outcome. The wider the gap between the two curves, the stronger that variable predicts winning.

team_stats <- matches |>
  group_by(matchId, teamId) |>
  summarise(teamVision = sum(visionScore), teamKills = sum(kills), .groups = "drop")

matches <- matches |>
  left_join(
    team_stats |>
      left_join(team_stats |> rename(enemyTeamId = teamId, enemyVision = teamVision, enemyKills = teamKills),
                by = "matchId", relationship = "many-to-many") |>
      filter(teamId != enemyTeamId) |>
      mutate(visionDiff = teamVision - enemyVision, killsDiff = teamKills - enemyKills) |>
      select(matchId, teamId, visionDiff, killsDiff),
    by = c("matchId", "teamId")
  )

ecdf_data <- bind_rows(
  matches |> distinct(matchId, teamId, .keep_all = TRUE) |> mutate(variable = "Gold Difference",          x = goldDiff * 1000),
  matches |> distinct(matchId, teamId, .keep_all = TRUE) |> mutate(variable = "Vision Score Difference",  x = visionDiff),
  matches |> distinct(matchId, teamId, .keep_all = TRUE) |> mutate(variable = "Kills Difference",         x = killsDiff)
) |>
  mutate(outcome = ifelse(win, "Win", "Loss"))

To test whether the separation between the Win and Loss gold difference curves is statistically significant, a two-sample Kolmogorov-Smirnov test was applied. The KS test is non-parametric and measures the maximum distance between two empirical cumulative distribution functions, making it well suited for this data since it requires no normality assumption.

A D statistic close to 1 indicates near-complete separation between the two distributions, while a value near 0 indicates they are indistinguishable. The near-zero p-value confirms the separation visible in the chart is not due to chance — teams with a positive gold difference win at a dramatically higher rate.

Personal Match History

Fetches all ranked solo/duo games from the Riot API (EUW) for the account tsilo#EUW.

library(httr)

riot_get <- function(url) {
  resp <- GET(url, add_headers("X-Riot-Token" = fromJSON(".kaggle/riot.json")$key))
  if (status_code(resp) != 200) stop("Riot API error: ", status_code(resp))
  content(resp, as = "parsed")
}

puuid <- riot_get("https://europe.api.riotgames.com/riot/account/v1/accounts/by-riot-id/tsilo/EUW")$puuid

match_ids <- {
  ids <- c(); start <- 0
  repeat {
    batch <- unlist(riot_get(paste0(
      "https://europe.api.riotgames.com/lol/match/v5/matches/by-puuid/", puuid,
      "/ids?queue=420&start=", start, "&count=100"
    )))
    ids <- c(ids, batch); Sys.sleep(1.2)
    if (length(batch) < 100) break
    start <- start + 100
  }
  ids
}

n <- length(match_ids)
raw_matches <- vector("list", n)
for (i in seq_len(n)) {
  raw_matches[[i]] <- tryCatch({
    d <- riot_get(paste0("https://europe.api.riotgames.com/lol/match/v5/matches/", match_ids[i]))
    Sys.sleep(0.1)
    p <- d$info$participants[[which(sapply(d$info$participants, \(x) x$puuid) == puuid)]]
    my_team   <- p$teamId
    my_gold   <- sum(sapply(Filter(\(x) x$teamId == my_team, d$info$participants), \(x) x$goldEarned))
    opp_gold  <- sum(sapply(Filter(\(x) x$teamId != my_team, d$info$participants), \(x) x$goldEarned))
    my_kills  <- sum(sapply(Filter(\(x) x$teamId == my_team, d$info$participants), \(x) x$kills))
    opp_kills <- sum(sapply(Filter(\(x) x$teamId != my_team, d$info$participants), \(x) x$kills))
    tibble(matchId = match_ids[i],
           date         = as.POSIXct(d$info$gameStartTimestamp / 1000, origin = "1970-01-01"),
           championName = p$championName, win = p$win,
           kills = p$kills, deaths = p$deaths, assists = p$assists,
           kda          = (p$kills + p$assists) / max(p$deaths, 1),
           goldEarned   = p$goldEarned, visionScore = p$visionScore,
           gameDuration = d$info$gameDuration / 60,
           goldDiff     = (my_gold - opp_gold) / 1000,
           killsDiff    = my_kills - opp_kills)
  }, error = function(e) NULL)
}
